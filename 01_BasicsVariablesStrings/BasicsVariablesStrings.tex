\bgsection{Karakterláncok}

\begin{frame}[fragile]{Literálok és indexelés}
  \begin{itemize}
    \item Idézőjelek: \texttt{"szöveg"} vagy \texttt{'szöveg'}.
    \item Első index: 0.
    \item Nincs külön karaktertípus: egy karakter is \texttt{str}.
  \end{itemize}
  \begin{block}{Példa}
\begin{verbatim}
text = "Portfolio"
print(text[0])
print(type(text[0]))
\end{verbatim}
  \end{block}
\end{frame}

\begin{frame}[fragile]{Keresés és szeletelés}
  \begin{itemize}
    \item Részkarakterlánc: \texttt{"HU" in isin}.
    \item Pozíció: \texttt{isin.find("HU")}.
    \item Szelet: \texttt{text[start:end:step]}.
  \end{itemize}
  \begin{block}{ISIN-példa}
\begin{verbatim}
isin = "HU0000403706"
start = isin.find("HU")
print("HU" in isin)
print(isin[start:start + 2])
\end{verbatim}
  \end{block}
\end{frame}

\begin{frame}[fragile]{String-metódusok}
  \begin{itemize}
    \item \texttt{len()}, \texttt{strip()}, \texttt{upper()}, \texttt{lower()}.
    \item \texttt{title()}, \texttt{replace()}.
  \end{itemize}
  \begin{block}{Példa}
\begin{verbatim}
name = "  danube VALUE fund  "
clean = name.strip()
print(clean.title())
print(len(clean))
\end{verbatim}
  \end{block}
\end{frame}

\begin{frame}[fragile]{Formázás}
  \begin{itemize}
    \item Összefűzés: \texttt{+}.
    \item Helyőrzők: \texttt{str.format()}.
    \item Modern forma: f-string.
    \item Számformátum: \texttt{:.2f}.
    \item Escape: \texttt{\textbackslash n}, \texttt{\textbackslash t}; raw string: \texttt{r"..."}.
  \end{itemize}
  \begin{block}{Kötvénykupon}
\begin{verbatim}
bond = "Danube Treasury 2031"
coupon = 0.0453
print(f"{bond}: {coupon * 100:.2f}%")
\end{verbatim}
  \end{block}
\end{frame}
